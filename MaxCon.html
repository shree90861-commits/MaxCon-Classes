    let currentExam = 'neet';

    let currentLevel = 'Class 11';

    let qIndex = 0;

    let score = 0;

    let attempted = 0;


    function updateRoadmap() {

      const d = data[currentExam];

      document.getElementById('roadmap-title').textContent = `1:1 Focus Plan for ${currentLevel}`;

      document.getElementById('roadmap-text').textContent = d.roadmaps[currentLevel];

      document.getElementById('level').value = currentLevel;

    }


    function renderQuestion() {

      const d = data[currentExam];

      const qObj = d.questions[qIndex];

      document.getElementById('quiz-progress').textContent = `Question ${qIndex + 1} of ${d.questions.length}`;

      document.getElementById('question').textContent = qObj.q;

      document.getElementById('feedback').textContent = 'Choose an answer to see the explanation.';

      document.getElementById('options').replaceChildren(...qObj.a.map((optText, i) => {

        const b = document.createElement('button');

        b.type = 'button';

        b.className = 'option';

        b.textContent = optText;

        b.onclick = () => {

          document.querySelectorAll('.option').forEach((o, j) => {

            o.disabled = true;

            if (j === qObj.correct) o.classList.add('correct');

          });

          attempted++;

          if (i === qObj.correct) {

            score++;

          } else {

            b.classList.add('wrong');

          }

          document.getElementById('score-tracker').textContent = `Score: ${score}/${attempted}`;

          document.getElementById('feedback').textContent = (i === qObj.correct ? '✓ Correct! ' : '✗ Not quite. ') + qObj.explain;

        };

        return b;

      }));

    }


    function setExam(key) {

      currentExam = key;

      qIndex = 0;

      const d = data[key];

      document.querySelectorAll('.tab').forEach(t => {

        const active = t.dataset.exam === key;

        t.classList.toggle('active', active);

        t.setAttribute('aria-selected', active);

      });

      document.getElementById('course-panel').setAttribute('aria-labelledby', 'tab-' + key);

      document.getElementById('course-tag').textContent = d.tag;

      document.getElementById('course-title').textContent = d.title;

      document.getElementById('course-desc').textContent = d.desc;

      document.getElementById('subjects').replaceChildren(...d.subjects.map(x => {

        const s = document.createElement('span');

        s.textContent = x;

        return s;

      }));

      document.getElementById('exam').value = d.exam;

      updateRoadmap();

      renderQuestion();

    }


    document.querySelectorAll('.tab').forEach(t => {

      t.onclick = () => setExam(t.dataset.exam);

    });


    document.querySelectorAll('.level-btn').forEach(btn => {

      btn.onclick = () => {

        currentLevel = btn.dataset.level;

        document.querySelectorAll('.level-btn').forEach(b => b.classList.toggle('active', b === btn));

      };

    });
